PhotoSwipe dimensions must match it. */}} {{ if in (slice "jpeg" "tiff") $resource.MediaType.SubType }} {{ with $resource.Exif }} {{ if in (slice 5 6 7 8) (int (.Tags.Orientation | default 1)) }} {{ $width = $resource.Height }} {{ $height = $resource.Width }} {{ end }} {{ end }} {{ end }} {{- $result = dict "Local" true "Resource" $resource "Permalink" $resource.RelPermalink "Height" $height "Width" $width }} {{- end -}} {{ else if not (in (slice "https" "http") $url.Scheme) }} {{/* Static images need dimensions to participate in Stack's native gallery. */}} {{ $staticPath := strings.TrimPrefix "/" .Image }} {{ $width := 0 }} {{ $height := 0 }} {{ $filePath := path.Join "static" $staticPath }} {{ if and (fileExists $filePath) (ne (lower (path.Ext $staticPath)) ".svg") }} {{ with try (images.Config $filePath) }} {{ with .Value }} {{ $width = .Width }} {{ $height = .Height }} {{ end }} {{ end }} {{ if in (slice ".jpg" ".jpeg" ".tif" ".tiff") (lower (path.Ext $staticPath)) }} {{ $staticResource := resources.FromString (printf "image-metadata/%s" $staticPath) (readFile $filePath) }} {{ with $staticResource.Exif }} {{ if in (slice 5 6 7 8) (int (.Tags.Orientation | default 1)) }} {{ $originalWidth := $width }} {{ $width = $height }} {{ $height = $originalWidth }} {{ end }} {{ end }} {{ end }} {{ end }} {{ $result = dict "Local" false "Resource" nil "Permalink" ($staticPath | relURL) "Height" $height "Width" $width }} {{ end }} {{ end }} {{ return $result }}
